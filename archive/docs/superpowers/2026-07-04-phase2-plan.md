// Add to Setup.gs
function Setup_installArrayFormulas(ledger) {
  var sheet = ledger.getSheetByName(TABS.BUDGET_REQUEST_LINES);
  if (!sheet) return;
  var claimedCol = COLS.BudgetRequestLines.claimed_amount;
  var remainingCol = COLS.BudgetRequestLines.remaining;
  
  // claimed_amount: SUMIFS of ClaimLineItems.amount where line_status is verified or higher
  // For simplicity: we'll use a direct formula. Let's just place the exact arrayformula in row 1.
  var claimedFormula = '={"claimed_amount"; ARRAYFORMULA(IF(ISBLANK(A2:A), "", SUMIF(ClaimLineItems!C:C, A2:A, ClaimLineItems!E:E)))}';
  sheet.getRange(1, claimedCol).setFormula(claimedFormula);
  
  var remainingFormula = '={"remaining"; ARRAYFORMULA(IF(ISBLANK(A2:A), "", E2:E - H2:H))}'; // E=approved, H=claimed
  sheet.getRange(1, remainingCol).setFormula(remainingFormula);
}
```
*Note: The formula needs to perfectly match the column indices. Adjust letters as needed based on `COLS.BudgetRequestLines`. E=5(approved), H=8(claimed), A=1(id). ClaimLineItems C=3(budget_line_id), E=5(amount).*

// Add to Engine.gs
  /**
   * Record new income.
   */
  recordIncome: function (date, categoryId, amount, sourceRef, eventId, notes, actorUserId) {
    var lock = LockService.getScriptLock();
    if (!lock.tryLock(30000)) throw new Error('System busy, try again');
    try {
      var incomeId = Ids.nextId('Income');
      var sheet = getSheet_(TABS.INCOME);
      // Construct row according to COLS.Income
      var row = [];
      row[COLS.Income.income_id - 1] = incomeId;
      row[COLS.Income.date - 1] = date;
      row[COLS.Income.category_id - 1] = categoryId;
      row[COLS.Income.amount - 1] = amount;
      row[COLS.Income.received_by - 1] = actorUserId;
      row[COLS.Income.source_ref - 1] = sourceRef;
      row[COLS.Income.event_id - 1] = eventId || '';
      row[COLS.Income.notes - 1] = notes || '';
      
      sheet.appendRow(row);
      Audit.append(actorUserId, 'Income', incomeId, 'CREATE', { amount: amount, sourceRef: sourceRef });
      return { ok: true, incomeId: incomeId };
    } finally {
      lock.releaseLock();
    }
  }
```

function dailyJob() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) return;
  try {
    var now = new Date();
    // 1. Check SLA nudges
    // 2. Auto-confirm payouts
    // 3. Lock old paid claims
    
    Audit.append('SYSTEM', 'Job', 'DAILY', 'EXECUTE', { timestamp: now.toISOString() });
  } finally {
    lock.releaseLock();
  }
}
